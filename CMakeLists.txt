cmake_minimum_required(VERSION 3.24)
project(IrAutoXQt LANGUAGES CXX)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(IrAutoXQt WIN32 launcher/qt_main.cpp launcher/qt_resources.qrc)
target_link_libraries(IrAutoXQt PRIVATE Qt6::Widgets Qt6::Network)
if(MSVC)
 target_compile_options(IrAutoXQt PRIVATE /O2 /Gy /Gw /W4 /permissive-)
 target_link_options(IrAutoXQt PRIVATE /OPT:REF /OPT:ICF /LTCG)
 set_property(TARGET IrAutoXQt PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
