cmake_minimum_required(VERSION 3.24)
project(IrAutoX VERSION 2.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "IrAutoX must be built for 32-bit Windows (Win32)")
endif()
if(NOT DEFINED SCITER_SDK_DIR OR NOT EXISTS "${SCITER_SDK_DIR}/include/sciter-x.h")
 message(FATAL_ERROR "Set SCITER_SDK_DIR to the Sciter SDK root")
endif()
include(FetchContent)
set(SCITER_DLL_FILE "${SCITER_SDK_DIR}/bin.win/x32/sciter.dll")
set(LOADING_GIF_FILE "${CMAKE_SOURCE_DIR}/LA.gif")
if(NOT EXISTS "${SCITER_DLL_FILE}" OR NOT EXISTS "${LOADING_GIF_FILE}")
 message(FATAL_ERROR "Embedded Sciter DLL or LA.gif is missing")
endif()
file(TO_NATIVE_PATH "${SCITER_DLL_FILE}" SCITER_DLL_RC_PATH)
file(TO_NATIVE_PATH "${LOADING_GIF_FILE}" LOADING_GIF_RC_PATH)
string(REPLACE "\\" "\\\\" SCITER_DLL_RC_PATH "${SCITER_DLL_RC_PATH}")
string(REPLACE "\\" "\\\\" LOADING_GIF_RC_PATH "${LOADING_GIF_RC_PATH}")
configure_file(launcher/resources.rc.in "${CMAKE_CURRENT_BINARY_DIR}/launcher_resources.rc" @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(nlohmann_json)
add_executable(IrAutoX WIN32 launcher/sciter_main.cpp launcher/discord_presence.cpp launcher/discord_presence.h "${CMAKE_CURRENT_BINARY_DIR}/launcher_resources.rc")
target_include_directories(IrAutoX PRIVATE "${SCITER_SDK_DIR}/include")
target_compile_definitions(IrAutoX PRIVATE UNICODE _UNICODE WINDOWS WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(IrAutoX PRIVATE /W4 /permissive- /O2 /Gy /Gw)
target_link_options(IrAutoX PRIVATE /OPT:REF /OPT:ICF /LTCG)
target_link_libraries(IrAutoX PRIVATE nlohmann_json::nlohmann_json winhttp comdlg32 shlwapi shell32 advapi32 ole32 dwmapi)
