cmake_minimum_required(VERSION 3.24)
project(IrAutoX VERSION 1.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "IrAutoX must be built for 32-bit Windows (Win32)")
endif()
if(NOT WEBVIEW2_ROOT)
 message(FATAL_ERROR "WEBVIEW2_ROOT must point to the Microsoft.Web.WebView2 NuGet package")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(UI_BUNDLE "${CMAKE_BINARY_DIR}/generated/ui_bundle.h")
add_custom_command(OUTPUT "${UI_BUNDLE}" COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/embed_ui.py" --html "${CMAKE_SOURCE_DIR}/ui/index.html" --tailwind "${CMAKE_BINARY_DIR}/tailwind.css" --output "${UI_BUNDLE}" DEPENDS "${CMAKE_SOURCE_DIR}/ui/index.html" "${CMAKE_SOURCE_DIR}/tools/embed_ui.py" "${CMAKE_BINARY_DIR}/tailwind.css" VERBATIM)
add_custom_target(IrAutoXUI DEPENDS "${UI_BUNDLE}")
add_executable(IrAutoX WIN32 launcher/main.cpp launcher/discord_presence.cpp launcher/discord_presence.h "${UI_BUNDLE}")
add_dependencies(IrAutoX IrAutoXUI)
target_include_directories(IrAutoX PRIVATE "${WEBVIEW2_ROOT}/build/native/include" "${CMAKE_BINARY_DIR}/generated")
target_link_directories(IrAutoX PRIVATE "${WEBVIEW2_ROOT}/build/native/x86")
target_compile_definitions(IrAutoX PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(IrAutoX PRIVATE /W4 /permissive- /O1 /Gy /Gw)
target_link_options(IrAutoX PRIVATE /OPT:REF /OPT:ICF /LTCG)
target_link_libraries(IrAutoX PRIVATE WebView2LoaderStatic.lib comctl32 comdlg32 shlwapi shell32 advapi32 ole32 dwmapi)
